예외 설계

class PaymentError(Exception): """이 패키지의 모든 결제 예외의 부모.""" class CardDeclinedError(PaymentError): def __init__(self, reason: str, *, code: str) -> None: super().__init__(f"card declined: {reason}") self.code = code class GatewayTimeoutError(PaymentError): pass try: resp = gateway.charge(card, amount) except httpx.TimeoutException as exc: raise GatewayTimeoutError("charge timed out") from exc
  • 호출자는 except PaymentError로 패키지 예외 전체를, 또는 구체 타입만 잡을 수 있음
  • 하위 라이브러리 예외(httpx)를 그대로 노출하면 구현이 API로 새어나감 → 변환
  • from exc: __cause__에 원인 저장 ("The above exception was the direct cause…")
  • from None: 원인 숨김 (내부 구현 디테일일 때만)
try: parse(row) except ValueError as exc: exc.add_note(f"while parsing line {lineno} of {path}") raise # 여러 작업의 실패를 한꺼번에 try: async with asyncio.TaskGroup() as tg: for u in urls: tg.create_task(fetch(u)) except* httpx.HTTPError as eg: for e in eg.exceptions: log.warning("fetch failed: %s", e)
안티패턴
• 흐름 제어용 예외 남발 (단, StopIteration처럼 관용적인 것은 OK)
• Exception("error") 같은 일반 예외 raise → 호출자가 구분 불가
• 예외 메시지에 비밀번호/토큰 포함
• 같은 예외를 여러 계층에서 로그 → 중복 로그 폭탄. "로그하거나 다시 던지거나, 둘 다 하지 않는다"

타입 시스템 활용: Protocol, TypedDict, Literal, overload

from typing import Protocol, TypedDict, Literal, Final, NotRequired, overload class SupportsClose(Protocol): # 구조적 서브타이핑 (덕 타이핑의 정적 버전) def close(self) -> None: ... def shutdown(*resources: SupportsClose) -> None: for r in resources: r.close() # 상속 없이도 close() 만 있으면 OK class UserPayload(TypedDict): id: int email: str nickname: NotRequired[str] Mode = Literal["r", "w", "a"] MAX_CONN: Final = 10 @overload def load(raw: str) -> dict: ... @overload def load(raw: bytes) -> dict: ... def load(raw: str | bytes) -> dict: ...
from typing import Self, TypeIs class Builder: def with_name(self, n: str) -> Self: # 서브클래스에서도 정확 self.name = n return self def is_str_list(v: list[object]) -> TypeIs[list[str]]: return all(isinstance(x, str) for x in v) class Stack[T]: def __init__(self) -> None: self._items: list[T] = [] def push(self, item: T) -> None: self._items.append(item) def pop(self) -> T: return self._items.pop()
리뷰 기준 ABC는 "is-a" 관계와 기본 구현 공유가 있을 때, Protocol은 "이 메서드만 있으면 된다"는 의존성 경계(테스트 더블 주입)에 적합합니다.

제너레이터와 지연 평가 파이프라인

def error_count(path): lines = open(path).readlines() errors = [l for l in lines if "ERROR" in l] return len(errors) def read_lines(path: Path) -> Iterator[str]: with path.open(encoding="utf-8") as f: yield from f def errors(lines: Iterable[str]) -> Iterator[str]: return (l for l in lines if "ERROR" in l) def error_count(path: Path) -> int: return sum(1 for _ in errors(read_lines(path)))
제너레이터 안의 with 제너레이터를 끝까지 소비하지 않으면 파일은 제너레이터가 GC될 때(또는 .close()) 닫힙니다. 즉시 정리가 중요하면 contextlib.closing(gen)이나 호출자가 리소스를 소유하는 구조로.
from itertools import islice, chain, groupby, pairwise, batched, accumulate first10 = list(islice(stream, 10)) all_rows = chain.from_iterable(pages) for a, b in pairwise([1, 4, 9]): ... # (1,4), (4,9) 3.10+ for chunk in batched(ids, 100): # 튜플 묶음 3.12+ bulk_insert(chunk) running = list(accumulate([1, 2, 3])) # [1, 3, 6] # groupby 는 "연속된" 같은 키만 묶는다 → 먼저 정렬! rows.sort(key=lambda r: r.dept) for dept, members in groupby(rows, key=lambda r: r.dept): print(dept, len(list(members)))

제너레이터의 return 값은 StopIteration.value로 전달되고 yield from의 결과가 됩니다. 제너레이터 안에서 StopIteration을 raise하면 3.7+에서 RuntimeError로 바뀝니다(PEP 479).

functools: 캐싱과 그 함정

from functools import cache, lru_cache, cached_property, partial, singledispatch, wraps @lru_cache(maxsize=1024) def get_rate(currency: str) -> Decimal: ... class Report: @cached_property # 인스턴스당 1회 계산 후 __dict__ 에 저장 def summary(self) -> Summary: ... @singledispatch def to_json(obj) -> object: raise TypeError(type(obj)) @to_json.register def _(obj: datetime) -> str: return obj.isoformat()
lru_cache 리뷰 체크리스트
  • 메서드에 @lru_cache: self가 캐시 키에 포함되어 인스턴스가 해제되지 않음 (메모리 누수). Ruff B019. → cached_property나 모듈 함수로
  • 인자가 hashable이어야 함 (list, dict 인자 → TypeError)
  • 가변 객체를 반환하면 호출자가 수정 시 캐시가 오염됨
  • maxsize=None(= @cache)은 무한 증가 → 입력 공간이 큰 경우 위험
  • 시간에 따라 바뀌는 값(환율, 설정)은 TTL이 없어 stale → 만료 정책 필요
  • cached_property는 __slots__ 클래스에선 동작 안 함, 스레드 락 없음(3.12에서 제거됨)

테스트 (pytest) 리뷰 포인트

import pytest @pytest.fixture def repo(tmp_path): r = SqliteRepo(tmp_path / "test.db") yield r # teardown 은 yield 뒤 r.close() @pytest.mark.parametrize( ("raw", "expected"), [ ("1,000", 1000), (" 42 ", 42), ("-7", -7), ], ) def test_parse_amount(raw, expected): assert parse_amount(raw) == expected def test_parse_amount_rejects_garbage(): with pytest.raises(ValueError, match="invalid amount"): parse_amount("abc") def test_uses_env(monkeypatch): monkeypatch.setenv("REGION", "kr") assert current_region() == "kr"
mock은 "사용되는 곳"을 patch # app/service.py from app.clients import fetch_user # 이름을 복사해 옴 # ❌ 원래 정의 위치를 patch → service 는 이미 복사한 참조를 사용 @patch("app.clients.fetch_user") # ✅ 조회되는 위치를 patch @patch("app.service.fetch_user", autospec=True)
  • autospec=True: 존재하지 않는 메서드/잘못된 시그니처 호출을 잡아냄 (assert_called_once_with 오타 같은 실수 방지)
  • mock 과다 = 구현에 결합된 테스트. 가능하면 가짜(fake) 구현이나 실제 경량 의존성(sqlite, tmp_path)
  • 하나의 테스트는 하나의 동작. 이름에 기대 동작을 담는다
  • 시간·난수·네트워크는 주입하거나 고정 (freezegun/time-machine, 시드)
  • 테스트 간 순서 의존·공유 상태 금지 (전역 캐시 초기화 fixture)
  • 경계값·실패 경로 테스트가 있는가? 커버리지 숫자보다 중요
