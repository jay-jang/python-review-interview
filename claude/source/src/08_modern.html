같은 코드, 옛날 스타일 vs 모던 스타일

from typing import Dict, List, Optional, TypeVar, Generic import os T = TypeVar("T") class Repo(Generic[T]): def __init__(self): self._items = {} # type: Dict[str, T] def get(self, key): # type: (str) -> Optional[T] return self._items.get(key) def load(path): if os.path.exists(path): f = open(path) text = f.read() f.close() cfg = dict(DEFAULTS) cfg.update(parse(text)) return cfg return None def kind(evt): if evt["type"] == "click" and "pos" in evt: x, y = evt["pos"] return "click %d,%d" % (x, y) elif evt["type"] == "key": return "key " + evt["key"]
from pathlib import Path class Repo[T]: def __init__(self) -> None: self._items: dict[str, T] = {} def get(self, key: str) -> T | None: return self._items.get(key) def load(path: Path) -> dict[str, str] | None: try: text = path.read_text(encoding="utf-8") except FileNotFoundError: return None return DEFAULTS | parse(text) def kind(evt: dict) -> str: match evt: case {"type": "click", "pos": (x, y)}: return f"click {x},{y}" case {"type": "key", "key": key}: return f"key {key}" case _: raise ValueError(f"unknown event: {evt!r}")

원본 kind()는 매칭 실패 시 암묵적으로 None을 반환했던 버그도 모던 버전에서 명시적 에러로 바뀌었습니다 — 리팩터링 제안은 동작 변화를 함께 설명하세요.
